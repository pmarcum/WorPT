%--- the spacing between columns: the table measured with the spacing set above and with none (its width grows evenly with

\def\[INSERTFILENAME]Width#1{% the table's width with spacing #1, into \dimen0
  \setlength{\tabcolsep}{#1}\def\SpaceBetweenColumns{#1}%
  \sbox\WorPTmeasureBox{\expandafter\tabular\expandafter{\[INSERTFILENAME]Columns}\[INSERTFILENAME]Body\endtabular}%
  \WorPTstraySpace{\[INSERTFILENAME]Columns}\dimen0=\dimexpr\wd\WorPTmeasureBox-\WorPTstray\relax}
\def\[INSERTFILENAME]Fit#1{% the spacing for a page #1 wide, into \WorPTcolumnSpace; false if even the least is too wide
  \ifdim\dimen4>#1\relax
    \global\WorPTcolumnSpace=\dimexpr\dimen2*\numexpr\dimexpr#1-\dimen6\relax\relax/\numexpr\dimexpr\dimen4-\dimen6\relax\relax\relax
    \ifdim\WorPTcolumnSpace<\dimen8 \global\WorPTcolumnSpace=\dimen8 \fi
    \[INSERTFILENAME]Width{\WorPTcolumnSpace}%
    \ifdim\dimen0>\dimexpr#1+0.1pt\relax\global\WorPTlandscapetrue\fi
  \fi}

  \[INSERTFILENAME]Width{\dimen2}\dimen4=\dimen0     % width with the spacing set above
  \[INSERTFILENAME]Width{0pt}\dimen6=\dimen0          % width with no spacing
  \global\WorPTcolumnSpace=\dimen2
  \ifLandScape\global\WorPTlandscapetrue\else\global\WorPTlandscapefalse\[INSERTFILENAME]Fit\textwidth\fi

    \ifWorPTlandscape\PackageWarningNoLine{worpt}{The travel table is wider than the page even in landscape mode}\fi

  \fi
  \endgroup}
