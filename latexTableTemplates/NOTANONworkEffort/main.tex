%--- table column widths
% The table is as wide as the page's text. The month columns are all as wide as their widest number or heading word;
% the Role column is as wide as its widest role; the Name column gets the width that is left (a long name takes a
% second line). If less than MinNameWidth would be left, the spacing between columns is made smaller, just enough,
% but not below MinSpaceBetweenColumns; if that is not enough, the table is set in landscape mode, which is wider
% (where the spacing is made smaller again if need be); if even that is not enough (five years, say), the headings over
% the months are set in SmallSubheadFont, which makes the month columns narrower.
\def\MinNameWidth{1.2in}              % narrowest Name column
\def\SpaceBetweenColumns{3pt}         % spacing between columns (the most it gets)
\def\MinSpaceBetweenColumns{1pt}      % the least spacing between columns

%---- table compactness controls
\def\SpaceBetweenRows{1}               % vertical compactness of rows

%--- the widths: the month columns as wide as their widest heading word or number; then the rest of the table measured
% without the names (and without the labels over the Name column: "Name", "Sum of work effort:", the comments), and the
% Name column given what is left of the page (measured with \sbox, as set in a table)
\def\[INSERTFILENAME]AtLeast#1#2{\ifdim#1<#2\global#1=#2\fi}
\def\[INSERTFILENAME]Setup{%
\renewcommand{\arraystretch}{\SpaceBetweenRows}% controls vertical row separation/spacing
\setlength{\tabcolsep}{\WorPTcolumnSpace}}     % the spacing between columns that fits the page (see the Layout)
\def\[INSERTFILENAME]Width#1{% the table's width without the names, with spacing #1 between columns, into \dimen0
  \global\WorPTtitleWidth=0pt \setlength{\tabcolsep}{#1}%
  \sbox\WorPTmeasureBox{\expandafter\tabular\expandafter{\[INSERTFILENAME]Columns}\[INSERTFILENAME]Body\endtabular}%
  \WorPTstraySpace{\[INSERTFILENAME]Columns}\dimen0=\dimexpr\wd\WorPTmeasureBox-\WorPTstray\relax}
\def\[INSERTFILENAME]Fit#1{% the spacing that leaves MinNameWidth for the names on a page #1 wide (no more than
  % SpaceBetweenColumns, no less than MinSpaceBetweenColumns), into \WorPTcolumnSpace; landscape mode if not even the
  % least spacing does; the Name column gets what is left
  \global\WorPTcolumnSpace=\dimen2
  \ifdim\dimexpr\dimen4+\MinNameWidth\relax>#1\relax
    \global\WorPTcolumnSpace=\dimexpr\dimen2*\numexpr\dimexpr#1-\MinNameWidth-\dimen6\relax\relax/\numexpr\dimexpr\dimen4-\dimen6\relax\relax\relax
    \ifdim\WorPTcolumnSpace<\dimen8 \global\WorPTcolumnSpace=\dimen8 \fi
  \fi
  \[INSERTFILENAME]Width{\WorPTcolumnSpace}%
  \ifdim\dimexpr\dimen0+\MinNameWidth\relax>\dimexpr#1+0.1pt\relax\global\WorPTlandscapetrue\fi
  \global\WorPTtitleWidth=\dimexpr#1-\dimen0\relax}
\def\[INSERTFILENAME]Measure{% the month columns' width; then the table's width with full spacing (\dimen4) and none (\dimen6)
  \global\WorPTnumberWidth=0pt
  \def\[INSERTFILENAME]Wide##1{\sbox0{##1}\[INSERTFILENAME]AtLeast\WorPTnumberWidth{\wd0}}%
  \[INSERTFILENAME]Wide{\SubheadFont NASA}\[INSERTFILENAME]Wide{\SubheadFont Support}\[INSERTFILENAME]Wide{\SubheadFont Total}%
  \[INSERTFILENAME]Wide{\SubheadFont Other}\[INSERTFILENAME]Wide{\SubheadFont Funded}\[INSERTFILENAME]Wide{\SubheadFont Projects}%
  \[INSERTFILENAME]Wide{\MonthsFormat{\WidestNumber}}\[INSERTFILENAME]Wide{\SumFormat{\WidestNumber}}%
  \[INSERTFILENAME]Width{\dimen2}\dimen4=\dimen0
  \[INSERTFILENAME]Width{0pt}\dimen6=\dimen0}
\def\[INSERTFILENAME]Layout{%
  \begingroup
  \csname @floatboxreset\endcsname  % measured as set in a table: some classes (aastex) use a smaller font there
  \renewcommand{\arraystretch}{\SpaceBetweenRows}%
  \global\WorPTspanWidth=0pt \WorPTmeasuringtrue
  \dimen2=\SpaceBetweenColumns\relax \dimen8=\MinSpaceBetweenColumns\relax
  \[INSERTFILENAME]Measure
  \ifLandScape\global\WorPTlandscapetrue\else\global\WorPTlandscapefalse\[INSERTFILENAME]Fit\textwidth\fi
  \ifWorPTlandscape
    \global\WorPTlandscapefalse\[INSERTFILENAME]Fit\textheight
    \ifWorPTlandscape                  % not even in landscape mode: smaller headings over the months
      \global\let\SubheadFont\SmallSubheadFont
      \[INSERTFILENAME]Measure\global\WorPTlandscapefalse\[INSERTFILENAME]Fit\textheight
    \fi
    \ifWorPTlandscape
      \global\WorPTtitleWidth=\MinNameWidth\relax
      \PackageWarningNoLine{worpt}{The work effort table is wider than the page even in landscape mode}%
    \fi
    \global\WorPTlandscapetrue
  \fi
  % the Comments row spans the table, inside its outer lines
  \global\WorPTspanWidth=\dimexpr\dimen0+\WorPTtitleWidth-2\arrayrulewidth-2\WorPTcolumnSpace\relax
  \endgroup}
