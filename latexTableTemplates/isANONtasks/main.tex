\newif\ifLandScape                 % establish a boolean variable

% ___________________________________________
%       Default formatting preferences
% ___________________________________________
\LandScapefalse                % landscape mode OFF, unless the table needs it (see MinTitleWidth); \LandScapetrue always uses it

%---- column widths
% The table is as wide as the page's text (in a two-column document it spans both columns). The "Task" column gets the
% width the other columns leave (a long title takes more than one line). If less than MinTitleWidth is left, the table
% is set in landscape mode, which is wider.
\def\MinTitleWidth{1in}        % narrowest "Task" column before switching to landscape mode
\def\StartFinishWidth{0.2in}  % width of start/finish columns
\def\LeadWidth{0.5in}          % width of "Task Lead" column
\def\ExpertiseWidth{1.5in}     % width of "Team Expertise" column (rightmost column)
\def\TaskWidth{\WorPTtitleWidth}% width of "Task" column (leftmost column): what is left (or give a width, such as 3.4in)

%---- table compactness controls
\def\SpaceBetweenRows{0.87}     % vertical compactness of rows
\def\SpaceBetweenColumns{2pt}  % spacing between columns


%--- aesthetic preferences
\def\SectionColor{Blue}        % task grouping banner color
\def\SectionFontColor{White}   % task grouping title font color
\def\YearFontColor{Red}        % font color under "Y" (year) columns in Start/Finish
\def\YearQuarterLineColor{lightgray} % vertical line color in year/quarter Start/Finish columns
\def\SectionFontstyle#1{\textbf{#1}} % boldface task grouping title
\def\ColumnLabelFontstyle#1{\textbf{#1}} % boldface all the column headers

%--- the table's columns
\def\[INSERTFILENAME]Columns{
  |>{\leavevmode}p{\TaskWidth}||       % title column
  >{\leavevmode}p{\StartFinishWidth}!{\color{\YearQuarterLineColor}\vrule} % Start/year
  >{\leavevmode}p{\StartFinishWidth}!{\color{\YearQuarterLineColor}\vrule}!{\color{\YearQuarterLineColor}\vrule} % Start/quarter
  >{\leavevmode}p{\StartFinishWidth}!{\color{\YearQuarterLineColor}\vrule} % Finish/year
  >{\leavevmode}p{\StartFinishWidth}|| % Finish/quarter
  >{\leavevmode}p{\LeadWidth}!{\color{\YearQuarterLineColor}\vrule} % Task Lead column
  >{\leavevmode}p{\ExpertiseWidth}|    % Expertise column
}

%--- the widths: the rest of the table measured with an empty "Task" column, and the "Task" column given what is left
\def\[INSERTFILENAME]Layout{%
  \begingroup
  \csname @floatboxreset\endcsname  % measured as set in a table: some classes (aastex) use a smaller font there
  \renewcommand{\arraystretch}{\SpaceBetweenRows}\setlength{\tabcolsep}{\SpaceBetweenColumns}%
  \def\TaskWidth{0pt}%
  \sbox\WorPTmeasureBox{\expandafter\tabular\expandafter{\[INSERTFILENAME]Columns}&&&&&&\\\endtabular}%
  \WorPTstraySpace{\[INSERTFILENAME]Columns}%
  \ifLandScape\global\WorPTlandscapetrue\else\global\WorPTlandscapefalse\fi
  \global\WorPTtitleWidth=\dimexpr\textwidth-\wd\WorPTmeasureBox+\WorPTstray\relax
  \ifdim\WorPTtitleWidth<\MinTitleWidth\relax\global\WorPTlandscapetrue\fi
  \ifWorPTlandscape\global\WorPTtitleWidth=\dimexpr\textheight-\wd\WorPTmeasureBox+\WorPTstray\relax\fi
  \ifdim\WorPTtitleWidth<\MinTitleWidth\relax
    \global\WorPTtitleWidth=\MinTitleWidth\relax
    \PackageWarningNoLine{worpt}{The tasks table is wider than the page even in landscape mode}%
  \fi
  \endgroup}
