%---- required input from user
\def\ProgramID{HST-xx-xxxxx (Cycle XX)} % user needs to fill in details

%---- column widths
% The table is as wide as the page's text (in a two-column document it spans both columns): the Role column gets the
% width the other columns leave.
\def\ContributorWidth{1.8in}      % Contributor column width
\def\PositionWidth{1.3in}         % Position column width
\def\FundedMemberWidth{0.22in}    % Funded(?) column width
\def\FteWidth{0.42in}             % FTE column width
\def\RoleWidth{\WorPTtitleWidth}  % Role column width: what is left (or give a width, such as 2.5in)


%---- table compactness controls
\def\SpaceBetweenRows{1}         % vertical compactness of rows
\def\SpaceBetweenColumns{1pt}    % spacing between columns

%--- aesthetic preferences
\def\HeaderColor{Blue}            % column heading color
\def\LabelColor{White}            % column heading font color
\def\LabelFontstyle#1{\textbf{#1}} % boldface column label
\def\VerticalLineColor{lightgray} % color of vertical lines

%--- internal definitions not to be used as user parameters
\def\AdminPIname{[INSERTADMINPINAME]}
\def\SciencePIname{[INSERTSCIENCEPINAME]}

%--- table preamble definition bundled into the "T" variable
\def\[INSERTFILENAME]Columns{
  |>{\raggedright\let\newline\\\arraybackslash\hspace{0pt}}p{\ContributorWidth}!{\color{\VerticalLineColor}\vrule} % Contributor column
  >{\raggedright\let\newline\\\arraybackslash\hspace{0pt}}p{\PositionWidth}!{\color{\VerticalLineColor}\vrule}     % Position column
  >{\raggedright\let\newline\\\arraybackslash\hspace{0pt}}p{\RoleWidth}!{\color{\VerticalLineColor}\vrule}         % Role column
  p{\FundedMemberWidth}!{\color{\VerticalLineColor}\vrule} % Funded(?) column
  p{\FteWidth}|                                            % FTE column
}

%--- the widths: the rest of the table measured with an empty Role column, and the Role column given what is left
\def\[INSERTFILENAME]Layout{%
  \begingroup
  \csname @floatboxreset\endcsname  % measured as set in a table: some classes (aastex) use a smaller font there
  \renewcommand{\arraystretch}{\SpaceBetweenRows}\setlength{\tabcolsep}{\SpaceBetweenColumns}%
  \def\RoleWidth{0pt}%
  \sbox\WorPTmeasureBox{\expandafter\tabular\expandafter{\[INSERTFILENAME]Columns}&&&&\\\endtabular}%
  \WorPTstraySpace{\[INSERTFILENAME]Columns}%
  \global\WorPTtitleWidth=\dimexpr\textwidth-\wd\WorPTmeasureBox+\WorPTstray\relax
  \ifdim\WorPTtitleWidth<0.5in   % (only on a page too narrow for the other columns)
    \global\WorPTtitleWidth=0.5in
    \PackageWarningNoLine{worpt}{The HST table is wider than the page}%
  \fi
  \endgroup}
